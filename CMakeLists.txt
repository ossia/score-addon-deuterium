if(WIN32 AND NOT MSVC)
  # On Windows with a non-MSVC compiler (no _MSC_VER), libgig's helper.h does
  #   #include "../win32/libgig_private.h"
  # but that Dev-C++-generated file does not exist in the submodule. Generate
  # a stand-in and provide it through the include path: quoted includes fall
  # back to the -I directories with the relative path applied, so a search
  # directory <X>/src resolves the include to <X>/win32/libgig_private.h.
  set(LIBGIG_COMPAT_DIR "${CMAKE_CURRENT_BINARY_DIR}/libgig-config")
  file(MAKE_DIRECTORY "${LIBGIG_COMPAT_DIR}/src")
  file(CONFIGURE
    OUTPUT "${LIBGIG_COMPAT_DIR}/win32/libgig_private.h"
    CONTENT [=[
#pragma once
/* Generated by score-addon-deuterium's CMakeLists as a stand-in for the
   Dev-C++-generated header expected by libgig's helper.h on non-MSVC
   Windows builds; helper.h only needs VER_STRING from it. */

/* helper.h defines PACKAGE and VERSION right after this include; drop the
   definitions passed on the command line to avoid redefinition warnings. */
#undef PACKAGE
#undef VERSION

#define VER_STRING "@LIBGIG_VERSION_MAJOR@.@LIBGIG_VERSION_MINOR@.@LIBGIG_VERSION_BUILD@"

/* helper.h only declares vasprintf() when HAVE_CONFIG_H or _MSC_VER is set,
   but helper.cpp defines it for every WIN32 build and helper.h's strPrint()
   calls it; mingw-flavored toolchains declare it in <stdio.h> only under
   _GNU_SOURCE. Mirror the declaration from helper.h's MSVC path. */
#include <stdarg.h>
int vasprintf(char** ret, const char* format, va_list arg);
]=]
    @ONLY)
  target_include_directories(${PROJECT_NAME} PRIVATE "${LIBGIG_COMPAT_DIR}/src")
endif()
